cmake_minimum_required(VERSION 3.21)
# The one version number lives in src/common/Version.h; the project version is read from it.
file(STRINGS src/common/Version.h TLR_VERSION_LINES REGEX "^#define TLR_VERSION_(MAJOR|MINOR|PATCH) ")
foreach(line ${TLR_VERSION_LINES})
    if(line MATCHES "TLR_VERSION_MAJOR ([0-9]+)")
        set(TLR_MAJOR ${CMAKE_MATCH_1})
    elseif(line MATCHES "TLR_VERSION_MINOR ([0-9]+)")
        set(TLR_MINOR ${CMAKE_MATCH_1})
    elseif(line MATCHES "TLR_VERSION_PATCH ([0-9]+)")
        set(TLR_PATCH ${CMAKE_MATCH_1})
    endif()
endforeach()
if(NOT DEFINED TLR_MAJOR OR NOT DEFINED TLR_MINOR OR NOT DEFINED TLR_PATCH)
    message(FATAL_ERROR "src/common/Version.h does not define TLR_VERSION_MAJOR, MINOR and PATCH")
endif()
project(ThunderheadLightningRod VERSION ${TLR_MAJOR}.${TLR_MINOR}.${TLR_PATCH} LANGUAGES C CXX)
message(STATUS "Thunderhead / Lightning Rod version ${PROJECT_VERSION} (from src/common/Version.h)")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(CMAKE_VS_PLATFORM_TOOLSET MATCHES "_xp$")
    set(TARGET_WINVER 0x0501)
else()
    set(TARGET_WINVER 0x0601)
endif()
add_compile_definitions(_WIN32_WINNT=${TARGET_WINVER} WINVER=${TARGET_WINVER} WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)

add_subdirectory(src/common)
add_subdirectory(src/LightningRod)
add_subdirectory(src/Thunderhead)
